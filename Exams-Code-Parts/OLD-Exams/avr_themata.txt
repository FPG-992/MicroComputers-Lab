(i)Διαβάζει κάποιο PORT και εμφανίζει το αντίστοιχο led και το αμέσως μεγαλύτερης αξίας μιας άλλης θύρας.Πχ αν δοθεί 0,όλα σβηστά,αν δοθεί 1 να εμφανίζει 0000 0011,αν δοθεί 7 1100 0000 κοκ.

reset:
		ldi r24 , low(RAMEND)	; initialize stack pointer
		out SPL , r24
		ldi r24 , high(RAMEND)
		out SPH , r24

		ser r24			; r24 = FF
		out DDRB , r24	;initialize PORTB for output
		clr r24
		out DDRA , r24	;initialize PINA for input
start_p:
		ldi r24,0x00
		out PORTB,r24

		in r28,PINA		;exoume thn timh toy value
		cpi r28,0x00
		breq start_p 
		cpi r28,0x08
		brge start_p

		dec r28		
		ldi r24,0x03
lop:	
		cpi r28,0x00
		breq display
		dec r28
		lsl r24
		jmp lop

display:
		out PORTB,r24
		jmp start_p

(ii)Από keypad 4x4,εισάγεται αριθμός χ ανήκει (0-D).Έξοδος στο PORTB,αν 0<=x<=4 ανάβουν τα bit 0-3 και εμφανίζει στην οθόνη LCD "BIT03",
αν 5<=x<=9 ανάβει led 4-7 και εμφανίζει στην LCD "BIT47".Όταν 10<=x<=13 ολα τα led αναβοσβήνουν με ορατό ρυθμό και εμφανίζεται στη LCD "EKTOS".

write_2_nibbles:
	push r24 				
	in r25 ,PIND 			
	andi r25 ,0x0f 			
	andi r24 ,0xf0 			
	add r24 ,r25 			
	out PORTD ,r24 			
	sbi PORTD ,PD3 		
	cbi PORTD ,PD3 			
	pop r24					
	swap r24 				
	andi r24 ,0xf0 			
	add r24 ,r25
	out PORTD ,r24
	sbi PORTD ,PD3 		
	cbi PORTD ,PD3
	ret
